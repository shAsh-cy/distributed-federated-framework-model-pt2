# Runtime dependencies. Every version pinned exactly.
#
# IMPORTANT: tensorflow-federated declares an exact pin `jaxlib==0.4.14`, and that
# release has been REMOVED from PyPI (the oldest jaxlib on PyPI is now 0.4.18).
# Installing from PyPI alone is therefore unsatisfiable at every TFF version, and an
# unpinned `pip install tensorflow-federated` silently back-tracks to the 2019
# placeholder 0.1.0, which contains none of the federated API.
#
# The --find-links line below restores jaxlib 0.4.14 from Google's historical index
# and is REQUIRED. Do not remove it.
#
# Platform note: TFF ships Linux-only wheels (manylinux_2_31_x86_64) and requires
# Python >=3.9,<3.12. Use Python 3.10 on Linux, WSL2, or the provided Docker images.
--find-links https://storage.googleapis.com/jax-releases/jax_releases.html
--extra-index-url https://download.pytorch.org/whl/cpu

# --- Federated learning stack ---
tensorflow==2.14.1
tensorflow-federated==0.87.0
tensorflow-privacy==0.9.0

# --- PyTorch clients (CPU build; the +cpu local tag needs the extra index) ---
# torch 2.0.1 is the newest torch that tolerates typing-extensions==4.5.*,
# which TFF pins exactly; torch >=2.1 requires typing-extensions>=4.8 and is
# therefore uninstallable beside tensorflow-federated. Verified empirically:
# this pin imports cleanly alongside tf 2.14.1 / tff 0.87.0 / numpy 1.25.2.
# The plain PyPI torch wheel would drag in ~2 GB of nvidia-* CUDA packages;
# the +cpu build from the pytorch index does not.
torch==2.0.1+cpu

# --- Privacy accounting ---
dp-accounting==0.4.3

# --- Transport (custom gRPC control plane) ---

# --- Core numerics / config ---

# --- Pinned transitively by TFF; recorded here so the resolution is reproducible ---
jax==0.4.14

typing-extensions==4.5.0

# --- Coordinator API (HTTP/WebSocket observability surface) ---
# The pydantic-v1 generation of the web stack, and not by preference:
# tensorflow-federated pins typing-extensions==4.5.*, pydantic v2 requires
# >=4.6, so fastapi must stay <0.100 (the last pydantic-v1 line). sqlalchemy
# is capped at 1.4.20 because google-vizier (a TFF dependency) declares
# sqlalchemy<=1.4.20; 2.x installs and appears to work, but the API shares a
# process with the live TFF DP aggregation path and an undeclared-conflict
# stack is not worth the risk. Verified empirically in the built image,
# including TFF DP aggregation running beside sqlalchemy's ORM.
